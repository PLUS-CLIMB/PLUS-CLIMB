// Load MODIS LST dataset
var modisLST = ee.ImageCollection("MODIS/006/MOD11A2")
  .select('LST_Day_1km');

// Function to confirm AOI
function confirmAOI() {
  var layers = drawingTools.layers();
  if (layers.length() === 0 || layers.get(0).getEeObject() === null) {
    print("No AOI found! Please draw a polygon first.");
    return;
  }

  var aoi = layers.get(0).getEeObject();
  
  // Display success message
  var aoiLabel = ui.Label('AOI Loaded Successfully!');
  panel.add(aoiLabel);
  
  // Perform handshake with MODIS LST data repository
  performHandshake(aoi);
}

// Function to perform handshake with MODIS LST data repository
function performHandshake(aoi) {
  // Check MODIS LST data availability
  var latestImage = modisLST.sort('system:time_start', false).first();
  var sample = latestImage.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: aoi,
    scale: 1000
  }).get('LST_Day_1km');
  
  sample.evaluate(function(value) {
    if (value !== null) {
      // Display handshake confirmation
      var handshakeLabel = ui.Label('Handshake with MODIS LST data repository successful!');
      panel.add(handshakeLabel);

      // Enable the Run Analysis button
      runButton.setDisabled(false);
    } else {
      print("No MODIS LST data found for this AOI. Try adjusting the location.");
    }
  });
}

// Function to run analysis
function runAnalysis() {
  var aoi = drawingTools.layers().get(0).getEeObject();
  var startDate = startDateInput.getValue();
  var endDate = endDateInput.getValue();
  
  var filtered = modisLST.filterDate(startDate, endDate).filterBounds(aoi)
    .map(function(image) {
      return image.multiply(0.02).subtract(273.15).copyProperties(image, ['system:time_start']);
    });
  var timeseries = filtered.mean().reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: aoi,
    scale: 1000,
    bestEffort: true
  });

  print("Analysis run from " + startDate + " to " + endDate);
  
  // Enable the Preview Graph, Preview Trendline, and Export Analysis buttons
  previewButton.setDisabled(false);
  trendlineButton.setDisabled(false);
  exportButton.setDisabled(false);
}

// Function to preview the graph
function previewGraph() {
  var aoi = drawingTools.layers().get(0).getEeObject();
  var startDate = startDateInput.getValue();
  var endDate = endDateInput.getValue();
  
  var filtered = modisLST.filterDate(startDate, endDate).filterBounds(aoi)
    .map(function(image) {
      return image.multiply(0.02).subtract(273.15).copyProperties(image, ['system:time_start']);
    });
  var timeseries = filtered.mean().reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: aoi,
    scale: 1000,
    bestEffort: true
  });

  var chart = ui.Chart.image.series(filtered, aoi, ee.Reducer.mean(), 1000)
      .setOptions({
        title: 'MODIS LST Time Series',
        vAxis: {title: 'LST (Celsius)'},
        hAxis: {title: 'Date'},
        lineWidth: 2,
        pointSize: 4,
        colors: ['orange']
      });

  panel.add(chart);
}

// Function to preview the trendline
function previewTrendline() {
  var aoi = drawingTools.layers().get(0).getEeObject();
  var startDate = startDateInput.getValue();
  var endDate = endDateInput.getValue();

  var filtered = modisLST.filterDate(startDate, endDate).filterBounds(aoi)
    .map(function(image) {
      return image.multiply(0.02).subtract(273.15).copyProperties(image, ['system:time_start']);
    });
  
  var trend = filtered.select('LST_Day_1km')
    .reduce(ee.Reducer.linearFit())
    .clip(aoi);

  var chart = ui.Chart.image.series(filtered, aoi, ee.Reducer.mean(), 1000)
      .setSeriesNames(['LST'])
      .setOptions({
        title: 'MODIS LST Trendline',
        vAxis: {title: 'LST (Celsius)'},
        hAxis: {title: 'Date'},
        lineWidth: 2,
        pointSize: 4,
        trendlines: {0: {}},
        colors: ['orange']
      });

  panel.add(chart);
}

// Function to export analysis
function exportAnalysis() {
  var folderPath = folderInput.getValue();
  var startDate = startDateInput.getValue();
  var endDate = endDateInput.getValue();
  var aoi = drawingTools.layers().get(0).getEeObject();

  // Define the export task
  var exportTask = Export.image.toDrive({
    image: modisLST.filterDate(startDate, endDate).mean().clip(aoi)
      .multiply(0.02).subtract(273.15),
    description: 'MODIS_LST_Time_Series_Export',
    folder: folderPath,
    fileNamePrefix: 'MODIS_LST_Time_Series',
    region: aoi,
    scale: 1000,
    fileFormat: 'GeoTIFF'
  });

  // Start the export task
  exportTask.start();
}
